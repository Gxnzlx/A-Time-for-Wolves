# Called after the lobby, when player character scopes are available.
# This mirrors the exposition timing used by CK3AGOT Vanilla for Robert's Rebellion.
on_game_start_after_lobby = {
	on_actions = {
		tfw_on_game_start_after_lobby
	}
}

tfw_on_game_start_after_lobby = {
	effect = {
		if = {
			limit = {
				game_start_date = 8303.12.31
			}
			every_player = {
				# The first bookmark has five featured actors.
				# Each receives a scenario-specific fullscreen war exposition,
				# following Robert's Rebellion's per-actor event structure.
				switch = {
					trigger = this

					character:Targaryen_9 = { trigger_event = atfw_bookmark_intro.0001 }
					character:Redwyne_2 = { trigger_event = atfw_bookmark_intro.0002 }
					character:Uller_6 = { trigger_event = atfw_bookmark_intro.0003 }
					character:Greyjoy_13 = { trigger_event = atfw_bookmark_intro.0004 }
					character:Lannister_6 = { trigger_event = atfw_bookmark_intro.0005 }
					character:Stark_7 = { trigger_event = atfw_bookmark_intro.0006 }
				}
			}

			# Jon's HBO chronology is a world storyline, not a player-only storyline.
			# If Jon is AI-controlled, start it independently of the selected player.
			if = {
				limit = {
					character:Stark_7 = {
						is_alive = yes
						is_ai = yes
						NOT = { has_character_flag = atfw_jon_story_bootstrapped }
					}
				}
				character:Stark_7 = {
					add_character_flag = atfw_jon_story_bootstrapped
					trigger_event = { id = atfw_weekly.1039 days = 7 }
				}
			}
		}
	}
}

tfw_north_holding_cost = {
	effect = {
		if = {
			limit = { current_year < 8305 }
			every_county_in_region = {
				region = world_westeros_the_north
				add_county_modifier = agot_holding_build_speed_north
			}
			every_county_in_region = {
				region = world_westeros_beyond_the_wall
				add_county_modifier = agot_holding_build_speed_btw
			}
			every_county_in_region = {
				region = world_westeros_the_wall
				add_county_modifier = agot_holding_build_speed_north
			}
		}
	}
}
