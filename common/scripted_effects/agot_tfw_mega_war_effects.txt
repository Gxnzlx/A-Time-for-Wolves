### Temporarily separate a historical Daenerys supporter for the 8303 Mega War.
# This mirrors Vanilla Mega War breakup behavior:
# 1. preserve the actual pre-war direct/top lieges;
# 2. become temporarily independent;
# 3. register in the rebel story so realm rebuild can restore vassalage.
atfw_8303_prepare_rebel_supporter = {
	$CHARACTER_SCOPE$ ?= {
		if = {
			limit = {
				is_landed = yes
				is_independent_ruler = no
			}

			if = {
				limit = {
					NOT = { has_variable = agot_pwl_direct }
				}
				agot_set_mw_pwl_variables = {
					SET_SCOPE = this
					REF_SCOPE = this
				}
			}

			agot_mw_become_independent = yes
		}

		scope:mw_rebel_story ?= {
			add_to_variable_list = {
				name = mw_rebel_supporter_list
				target = prev
			}
		}

		every_vassal_or_below = {
			limit = {
				highest_held_title_tier > tier_barony
				agot_is_high_septon = no
				NOT = { primary_title = title:d_kingsguard }
			}
			scope:mw_rebel_story ?= {
				add_to_variable_list = {
					name = mw_rebel_supporter_list
					target = prev
				}
			}
		}
	}
}
