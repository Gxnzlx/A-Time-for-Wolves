# Called in CB outcome scope. All delayed data is captured before AGOT cleans
# the Mega War lists; no expired war scope is used by the settlement event.
atfw_conquest_snapshot_effect = {
 scope:atfw_conquest_winner = { add_character_flag = atfw_conquest_victory_pending }
 # Capture the defeated court/guard before peace transfers dismiss them.
 scope:attacker = { every_courtier = { add_to_list = atfw_conquest_daenerys_court } }
 every_living_character = {
  limit = { has_trait = kingsguard }
  add_to_list = atfw_conquest_royal_guard
 }
 title:d_kingsguard.holder ?= { add_to_list = atfw_conquest_royal_guard }

 scope:attacker = {
  every_owned_story = {
   limit = {
    story_type = story_agot_mw_rebel
    exists = var:mw_crown_story_var
    var:mw_crown_story_var.story_owner = scope:defender
   }
   every_in_list = {
    variable = mw_rebel_supporter_list
    add_to_list = atfw_conquest_participants
   }
  }
 }
 # Actual combatants include rulers who joined after the initial setup.
 every_in_list = { list = attackers add_to_list = atfw_conquest_participants }
 every_in_list = { list = defenders add_to_list = atfw_conquest_participants }
 scope:atfw_conquest_loser = {
  every_vassal = { add_to_list = atfw_conquest_former_vassals }
 }
 # Include temporarily detached direct vassals even if they stayed neutral.
 # Independent neutral rulers with no former liege are deliberately excluded.
 every_ruler = {
  limit = {
   exists = var:agot_pwl_direct
   var:agot_pwl_direct = scope:atfw_conquest_loser
  }
  add_to_list = atfw_conquest_former_vassals
 }
 debug_log = "ATFW conquest: captured war participants and former vassals"
}

atfw_conquest_take_war_targets_effect = {
 create_title_and_vassal_change = {
  type = usurped
  save_scope_as = atfw_conquest_change
  add_claim_on_loss = no
 }
 every_in_list = {
  list = target_titles
  limit = { exists = holder holder = scope:atfw_conquest_loser }
  change_title_holder_include_vassals = {
   holder = scope:atfw_conquest_winner
   change = scope:atfw_conquest_change
  }
 }
 resolve_title_and_vassal_change = scope:atfw_conquest_change

 scope:atfw_conquest_winner = {
 if = {
  limit = { has_title = title:h_the_iron_throne }
  set_primary_title_to = title:h_the_iron_throne
  if = { limit = { has_title = title:c_kings_landing } set_realm_capital = title:c_kings_landing }
 }
 }
}

# Character scope. Restore a surviving intermediate lord when possible;
# otherwise the conquered/committed independent realm joins the victor.
atfw_conquest_attach_ruler_effect = {
 if = {
  limit = {
   is_alive = yes
   is_landed = yes
   NOT = { this = scope:atfw_conquest_winner }
   NOT = { this = scope:atfw_conquest_loser }
   is_independent_ruler = yes
   highest_held_title_tier < scope:atfw_conquest_winner.highest_held_title_tier
  }
  if = {
   limit = { exists = var:agot_pwl_direct }
   agot_reassign_pwl_direct_effect = yes
  }
  if = {
   limit = { is_independent_ruler = yes }
   create_title_and_vassal_change = {
    type = swear_fealty
    save_scope_as = atfw_conquest_vassal_change
    add_claim_on_loss = no
   }
   change_liege = {
    liege = scope:atfw_conquest_winner
    change = scope:atfw_conquest_vassal_change
   }
   resolve_title_and_vassal_change = scope:atfw_conquest_vassal_change
   # Any later Mega War cleanup must preserve this settlement.
   set_variable = { name = agot_pwl_direct value = scope:atfw_conquest_winner }
   set_variable = { name = agot_pwl_top value = scope:atfw_conquest_winner }
  }
 }
}

atfw_conquest_settle_domains_effect = {
 # Redirect stale links before the defeated ruler loses the remaining titles.
 every_ruler = {
  if = {
   limit = { exists = var:agot_pwl_direct var:agot_pwl_direct = scope:atfw_conquest_loser }
   set_variable = { name = agot_pwl_direct value = scope:atfw_conquest_winner }
  }
  if = {
   limit = { exists = var:agot_pwl_top var:agot_pwl_top = scope:atfw_conquest_loser }
   set_variable = { name = agot_pwl_top value = scope:atfw_conquest_winner }
  }
 }
 if = {
  limit = { this = character:Lannister_6 }
  atfw_conquest_destroy_daenerys_domain_effect = yes
 }
 else = {
 create_title_and_vassal_change = {
  type = usurped
  save_scope_as = atfw_conquest_change
  add_claim_on_loss = no
 }
 scope:atfw_conquest_loser = {
  every_held_title = {
   limit = { NOT = { AND = { tier = tier_barony exists = title_province title_province ?= { is_county_capital = yes } } } }
   change_title_holder_include_vassals = {
    holder = scope:atfw_conquest_winner
    change = scope:atfw_conquest_change
   }
  }
 }
 resolve_title_and_vassal_change = scope:atfw_conquest_change
 }
 every_in_list = {
  list = atfw_conquest_former_vassals
  atfw_conquest_attach_ruler_effect = yes
 }
 every_in_list = {
  list = atfw_conquest_participants
  # Foreign military allies are not conquered merely for joining a war.
  limit = { is_alive = yes is_landed = yes exists = capital_county capital_county ?= { target_is_de_jure_liege_or_above = title:h_the_iron_throne } }
  atfw_conquest_attach_ruler_effect = yes
 }
 # All transfers and realm reconstruction precede the executions.
 if = {
  limit = { this = character:Lannister_6 }
  every_in_list = {
   list = atfw_conquest_daenerys_court
   limit = { is_alive = yes NOT = { this = scope:atfw_conquest_winner } }
   death = { death_reason = death_execution killer = scope:atfw_conquest_winner }
  }
  character:Stark_7 ?= {
   if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:atfw_conquest_winner } }
  }
 }
 else = {
  every_in_list = {
   list = atfw_conquest_royal_guard
   limit = { is_alive = yes NOT = { this = scope:atfw_conquest_winner } }
   death = { death_reason = death_execution killer = scope:atfw_conquest_winner }
  }
 }
 scope:atfw_conquest_loser = {
  if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:atfw_conquest_winner } }
 }
 debug_log = "ATFW conquest: victory option resolved"
}

atfw_conquest_destroy_daenerys_domain_effect = {
 scope:atfw_conquest_winner = { save_scope_as = atfw_driftmark_recipient }
 if = {
  limit = { exists = title:d_driftmark.holder title:d_driftmark.holder = { is_alive = yes } }
  title:d_driftmark.holder = { save_scope_as = atfw_driftmark_recipient }
 }
 else_if = {
  limit = { exists = title:c_driftmark.holder title:c_driftmark.holder = { is_alive = yes } }
  title:c_driftmark.holder = { save_scope_as = atfw_driftmark_recipient }
 }
 # Counties and baronies cannot be destroyed. Preserve their ownership before
 # dissolving the defeated queen's duchies, kingdoms and higher titles.
 scope:atfw_conquest_loser = {
  every_held_title = {
   limit = { tier > tier_county }
   add_to_list = atfw_conquest_titles_to_destroy
  }
 }
 create_title_and_vassal_change = { type = usurped save_scope_as = atfw_conquest_change add_claim_on_loss = no }
 scope:atfw_conquest_loser = {
  every_held_title = {
   limit = { tier <= tier_county NOT = { AND = { tier = tier_barony exists = title_province title_province ?= { is_county_capital = yes } } } }
   if = {
    limit = { OR = { this = title:c_dragonstone county = title:c_dragonstone } }
    change_title_holder_include_vassals = { holder = scope:atfw_driftmark_recipient change = scope:atfw_conquest_change }
   }
   else = {
    change_title_holder_include_vassals = { holder = scope:atfw_conquest_winner change = scope:atfw_conquest_change }
   }
  }
 }
 resolve_title_and_vassal_change = scope:atfw_conquest_change
 every_in_list = {
  list = atfw_conquest_titles_to_destroy
  scope:atfw_conquest_winner = { destroy_title = prev }
 }
}

atfw_conquest_resolve_victory_option_effect = {
 if = {
  limit = { NOT = { has_character_flag = atfw_conquest_domains_settled } }
  add_character_flag = atfw_conquest_domains_settled
  atfw_conquest_settle_domains_effect = yes
  remove_character_flag = atfw_conquest_victory_pending
  # Defer ATFW's existing grants until the player has confiscated the domain.
  if = {
   limit = { this = character:Targaryen_9 has_character_flag = targ_host_invasion has_title = title:h_the_iron_throne }
   trigger_event = { id = agot_tfw_five_queens.0004 days = 1 }
  }
 }
}
