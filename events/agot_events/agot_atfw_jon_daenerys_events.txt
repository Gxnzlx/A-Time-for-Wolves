# Jon Snow has physically reached Dragonstone. This event is fired directly
# from the active destination phase of activity_atfw_journey_to_dragonstone.

	type = activity_event
	title = atfw_jon_daenerys.0001.t
	desc = atfw_jon_daenerys.0001.desc
	theme = diplomacy

	immediate = {
		remove_character_flag = atfw_journey_to_dragonstone_arrived
		character:Targaryen_9 = { save_scope_as = daenerys }
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = {
		name = atfw_jon_daenerys.0001.a
		scope:daenerys = {
			trigger_event = { id = atfw_jon_daenerys.0002 days = 1 }
		}
		ai_chance = { base = 100 }
	}
}

# Daenerys decides whether political understanding will become a dynastic union.
atfw_jon_daenerys.0004 = {
	type = character_event
	title = atfw_jon_daenerys.0004.t
	desc = atfw_jon_daenerys.0004.desc
	theme = marriage

	trigger = {
		is_alive = yes
		character:Stark_7 = {
			is_alive = yes
			involved_activity ?= {
				has_activity_type = activity_atfw_journey_to_dragonstone
			}
		}
	}

	immediate = {
		character:Stark_7 = { save_scope_as = jon }
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:jon
		animation = personality_rational
	}

	option = {
		name = atfw_jon_daenerys.0004.a

		# Acceptance is only possible while both remain free to marry.
		trigger = {
			is_married = no
			is_betrothed = no
			has_trait = refusing_marriage
			scope:jon = {
				is_married = no
				is_betrothed = no
				has_trait = refusing_marriage
			}
		}

		remove_trait = refusing_marriage
		scope:jon = {
			remove_trait = refusing_marriage
			create_betrothal = root
		}

		# The personal union is followed by the political settlement established
		# for this ATFW scenario: the North and Vale swear directly to Daenerys.
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = jon_daenerys_union_change
			add_claim_on_loss = no
		}

		title:e_the_north.holder ?= {
			change_liege = {
				liege = root
				change = scope:jon_daenerys_union_change
			}
		}

		title:e_the_vale.holder ?= {
			change_liege = {
				liege = root
				change = scope:jon_daenerys_union_change
			}
		}

		resolve_title_and_vassal_change = scope:jon_daenerys_union_change

		# Jon was independent before this union. Any PWL data left from the
		# bookmark Mega War setup is stale and must not survive his voluntary
		# vassalization to Daenerys.
		scope:jon = {
			remove_variable = agot_pwl_direct
			remove_variable = agot_pwl_top
		}

		# If Daenerys's conquest is still active, Jon joins her as a
		# co-belligerent WITHOUT breaking the vassalage just created above.
		# His new liege relationship with Daenerys is intentional and must not
		# be treated as temporary Mega War breakup data.
		random_owned_story = {
			limit = {
				story_type = story_agot_mw_rebel
				story_owner = root
			}
			save_scope_as = jon_daenerys_mw_rebel_story
		}

		random_character_war = {
			limit = {
				using_cb = atfw_daenerys_conquest_war
				primary_attacker = root
				primary_defender = title:h_the_iron_throne.holder
			}
			save_scope_as = jon_daenerys_conquest_war
		}

		if = {
			limit = {
				exists = scope:jon_daenerys_mw_rebel_story
				exists = scope:jon_daenerys_conquest_war
			}

			scope:jon_daenerys_mw_rebel_story.var:mw_crown_story_var ?= {
				save_scope_as = jon_daenerys_mw_crown_story
			}

			# Jon was initially registered as a neutral independent ruler.
			# Clear that old stance and any stale PWL data before moving the
			# North to Daenerys's side.
			scope:jon = {
				remove_variable = agot_pwl_direct
				remove_variable = agot_pwl_top

				scope:jon_daenerys_mw_crown_story ?= {
					remove_list_variable = {
						name = mw_neutral_list
						target = prev
					}
					remove_list_variable = {
						name = mw_loyalist_list
						target = prev
					}
				}

				scope:jon_daenerys_mw_rebel_story = {
					add_to_variable_list = {
						name = mw_rebel_supporter_list
						target = prev
					}
				}

				every_vassal_or_below = {
					limit = {
						highest_held_title_tier > tier_barony
						agot_is_high_septon = no
						NOT = { primary_title = title:d_kingsguard }
					}

					scope:jon_daenerys_mw_crown_story ?= {
						remove_list_variable = {
							name = mw_neutral_list
							target = prev
						}
						remove_list_variable = {
							name = mw_loyalist_list
							target = prev
						}
					}

					scope:jon_daenerys_mw_rebel_story = {
						add_to_variable_list = {
							name = mw_rebel_supporter_list
							target = prev
						}
					}
				}
			}

			scope:jon_daenerys_conquest_war = {
				add_attacker = scope:jon
			}
		}

		# The diplomatic visit has reached its successful conclusion.
		scope:jon = {
			remove_character_flag = atfw_journey_to_dragonstone_travelling
			remove_character_flag = atfw_journey_to_dragonstone_arrived
			involved_activity ?= {
				progress_activity_phase_after = { days = 1 }
			}
		}

		ai_chance = { base = 80 }
	}

	option = {
		name = atfw_jon_daenerys.0004.b

		# Daenerys keeps refusing_marriage; no betrothal or realm transfer occurs.
		scope:jon = {
			trigger_event = { id = atfw_jon_daenerys.0006 days = 1 }
		}

		ai_chance = { base = 20 }
	}
}
