# Part I of Jon Snow's 8303 storyline.
# This chain is deliberately Jon-only. It establishes the immediate
# post-Battle-of-the-Bastards situation before the existing Dragonstone journey.

# Bran reaches Winterfell before the northern council.
atfw_jon_north.0001 = {
	type = character_event
	title = atfw_jon_north.0001.t
	desc = atfw_jon_north.0001.desc
	theme = family
	override_background = {
		reference = agot_pk_north
	}

	immediate = {
		# Stark_Dummy_10 now represents Bran in the active timeline. Detach the
		# vanished original so only the continuation appears among Ned and
		# Catelyn's children in the family tree.
		character:Stark_10 = {
			set_father = character:Unused_1
			set_mother = character:Unused_2
		}

		character:Stark_Dummy_10 = {
			change_first_name = Brandon
			set_father = character:Stark_3
			set_mother = character:Tully_4
			set_house = character:Stark_3.house
			set_location = root.capital_province
			save_scope_as = bran
		}
		character:Stark_Dummy_10 = {
			if = {
				limit = { NOT = { is_courtier_of = root } }
				root = { add_courtier = prev }
			}
			return_to_court = yes
		}
	}

	left_portrait = {
		character = root
		animation = shock
	}

	right_portrait = {
		character = scope:bran
		animation = personality_rational
	}

	option = {
		name = atfw_jon_north.0001.a
		trigger_event = { id = atfw_jon_north.0002 days = 10 }
	}
}

# A private gathering reveals Jon's true parentage.
# This adapts CK3AGOT's existing disputed-heritage/royal-bastard secret
# mechanics without publicly exposing the secret or changing Jon's public house.
atfw_jon_north.0005 = {
	type = character_event
	title = atfw_jon_north.0005.t
	desc = atfw_jon_north.0005.desc
	theme = family

	trigger = {
		has_character_flag = atfw_jon_proclaimed_king_in_the_north
		has_character_flag = atfw_jon_war_council_complete
		has_character_flag = atfw_jon_northern_feast_completed
		NOT = { has_character_flag = atfw_jon_parentage_scene_fired }
	}

	override_background = study

	immediate = {
		# Set first: queued duplicate copies will no longer pass this event's trigger.
		add_character_flag = atfw_jon_parentage_scene_fired
		remove_character_flag = atfw_jon_parentage_scene_scheduled

		character:Stark_Dummy_8 = { save_scope_as = sansa }
		character:Stark_Dummy_10 = { save_scope_as = bran }
		character:Tarly_2 = { save_scope_as = sam }
		character:Reed_1 = { save_scope_as = howland }

		character:Reed_2 = {
			set_location = root.capital_province
			save_scope_as = meera

			if = {
				limit = {
					NOT = { is_courtier_of = root }
				}
				root = { add_courtier = prev }
			}
		}

		if = {
			limit = {
				knows_self_royal_bastard_secret = no
			}
			scope:howland = {
				random_known_secret = {
					limit = {
						secret_type = secret_agot_disputed_heritage
						secret_target = root
					}
					reveal_to = root
				}
			}
		}

		if = {
			limit = {
				knows_self_royal_bastard_secret = no
			}
			add_secret = {
				type = secret_agot_disputed_heritage
				target = root
			}
		}

		add_character_flag = atfw_jon_knows_true_parentage
	}

	left_portrait = {
		character = scope:bran
		animation = personality_rational
	}

	right_portrait = {
		character = scope:sansa
		animation = shock
	}

	lower_left_portrait = {
		character = scope:sam
		animation = personality_honorable
	}

	lower_center_portrait = {
		character = scope:howland
		animation = personality_rational
	}

	lower_right_portrait = {
		character = scope:meera
		animation = personality_bold
	}

	option = {
		name = atfw_jon_north.0005.a
		add_character_flag = atfw_jon_hides_true_parentage
		trigger_event = {
			id = atfw_jon_north.0006
			days = 12
		}
	}
}

# Daenerys decides whether political understanding will become a dynastic union.
atfw_jon_daenerys.0004 = {
	type = character_event
	title = atfw_jon_daenerys.0004.t
	desc = atfw_jon_daenerys.0004.desc
	theme = marriage

	trigger = {
		is_alive = yes
		character:Stark_7 = {
			is_alive = yes
			involved_activity ?= {
				has_activity_type = activity_atfw_journey_to_dragonstone
			}
		}
	}

	immediate = {
		character:Stark_7 = { save_scope_as = jon }
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:jon
		animation = personality_rational
	}

	option = {
		name = atfw_jon_daenerys.0004.a

		# Acceptance is only possible while both remain free to marry.
		trigger = {
			is_married = no
			is_betrothed = no
			has_trait = refusing_marriage
			scope:jon = {
				is_married = no
				is_betrothed = no
				has_trait = refusing_marriage
			}
		}

		remove_trait = refusing_marriage
		scope:jon = {
			remove_trait = refusing_marriage
			create_betrothal = root
		}

		# The personal union is followed by the political settlement established
		# for this ATFW scenario: the North and Vale swear directly to Daenerys.
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = jon_daenerys_union_change
			add_claim_on_loss = no
		}

		title:e_the_north.holder ?= {
			change_liege = {
				liege = root
				change = scope:jon_daenerys_union_change
			}
		}

		title:e_the_vale.holder ?= {
			change_liege = {
				liege = root
				change = scope:jon_daenerys_union_change
			}
		}

		resolve_title_and_vassal_change = scope:jon_daenerys_union_change

		# Jon was independent before this union. Any PWL data left from the
		# bookmark Mega War setup is stale and must not survive his voluntary
		# vassalization to Daenerys.
		scope:jon = {
			remove_variable = agot_pwl_direct
			remove_variable = agot_pwl_top
		}

		# If Daenerys's conquest is still active, Jon joins her as a
		# co-belligerent WITHOUT breaking the vassalage just created above.
		# His new liege relationship with Daenerys is intentional and must not
		# be treated as temporary Mega War breakup data.
		random_owned_story = {
			limit = {
				story_type = story_agot_mw_rebel
				story_owner = root
			}
			save_scope_as = jon_daenerys_mw_rebel_story
		}

		random_character_war = {
			limit = {
				using_cb = atfw_daenerys_conquest_war
				primary_attacker = root
				primary_defender = title:h_the_iron_throne.holder
			}
			save_scope_as = jon_daenerys_conquest_war
		}

		if = {
			limit = {
				exists = scope:jon_daenerys_mw_rebel_story
				exists = scope:jon_daenerys_conquest_war
			}

			scope:jon_daenerys_mw_rebel_story.var:mw_crown_story_var ?= {
				save_scope_as = jon_daenerys_mw_crown_story
			}

			# Jon was initially registered as a neutral independent ruler.
			# Clear that old stance and any stale PWL data before moving the
			# North to Daenerys's side.
			scope:jon = {
				remove_variable = agot_pwl_direct
				remove_variable = agot_pwl_top

				scope:jon_daenerys_mw_crown_story ?= {
					remove_list_variable = {
						name = mw_neutral_list
						target = prev
					}
					remove_list_variable = {
						name = mw_loyalist_list
						target = prev
					}
				}

				scope:jon_daenerys_mw_rebel_story = {
					add_to_variable_list = {
						name = mw_rebel_supporter_list
						target = prev
					}
				}

				every_vassal_or_below = {
					limit = {
						highest_held_title_tier > tier_barony
						agot_is_high_septon = no
						NOT = { primary_title = title:d_kingsguard }
					}

					scope:jon_daenerys_mw_crown_story ?= {
						remove_list_variable = {
							name = mw_neutral_list
							target = prev
						}
						remove_list_variable = {
							name = mw_loyalist_list
							target = prev
						}
					}

					scope:jon_daenerys_mw_rebel_story = {
						add_to_variable_list = {
							name = mw_rebel_supporter_list
							target = prev
						}
					}
				}
			}

			scope:jon_daenerys_conquest_war = {
				add_attacker = scope:jon
			}
		}

		# The diplomatic visit has reached its successful conclusion.
		scope:jon = {
			remove_character_flag = atfw_journey_to_dragonstone_travelling
			remove_character_flag = atfw_journey_to_dragonstone_arrived
			involved_activity ?= {
				progress_activity_phase_after = { days = 1 }
			}
		}

		ai_chance = { base = 80 }
	}

	option = {
		name = atfw_jon_daenerys.0004.b

		# Daenerys keeps refusing_marriage; no betrothal or realm transfer occurs.
		scope:jon = {
			trigger_event = { id = atfw_jon_daenerys.0006 days = 1 }
		}

		ai_chance = { base = 20 }
	}
}
