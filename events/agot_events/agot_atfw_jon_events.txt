# The lords renew their loyalty and proclaim Jon King in the North.
# This remains ONE event window inside the same feast activity.
atfw_jon_north.0003 = {
	type = activity_event
	title = atfw_jon_north.0003.t
	desc = atfw_jon_north.0003.desc
	theme = feast_activity
	override_background = {
		reference = agot_winterfell_throneroom
	}

	trigger = {
		this = character:Stark_7
		NOT = { has_character_flag = atfw_jon_proclaimed_king_in_the_north }
	}

	immediate = {
		character:Mormont_9 = { save_scope_as = lyanna_mormont }
		character:Stark_Dummy_8 = { save_scope_as = sansa }

		# General loyalty declarations happen inside this single feast scene;
		# no per-lord event windows are opened.
		scope:activity = {
			every_attending_character = {
				limit = {
					this != scope:host
					is_landed = yes
					OR = {
						is_vassal_of = scope:host
						target_is_vassal_or_below = scope:host
					}
					NOT = { has_character_flag = atfw_northern_banners_fealty_renewed }
				}
				add_character_flag = atfw_northern_banners_fealty_renewed
				add_opinion = {
					target = scope:host
					modifier = grateful_opinion
					opinion = 15
				}
			}
		}
	}

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:lyanna_mormont
		animation = personality_bold
	}
	lower_left_portrait = {
		character = scope:sansa
		animation = personality_rational
	}

	option = {
		name = atfw_jon_north.0003.a
		add_prestige = major_prestige_gain
		add_character_flag = atfw_jon_proclaimed_king_in_the_north
	}
}

# The ship scene: "Dany"; she commits to the war, Jon recognizes her as queen.
atfw_jon_daenerys.0015 = {
	type = character_event
	title = atfw_jon_daenerys.0015.t
	desc = atfw_jon_daenerys.0015.desc
	theme = diplomacy
	override_background = {
		reference = AGOT_rowboat_night_moonlight
	}

	immediate = {
		character:Targaryen_9 = { save_scope_as = daenerys }
	}
	right_portrait = {
		character = scope:daenerys
		animation = personality_rational
	}

	option = {
		name = atfw_jon_daenerys.0015.a
		add_character_flag = atfw_jon_swore_fealty_to_daenerys
		scope:daenerys = { add_character_flag = atfw_daenerys_committed_to_great_war }
		set_relation_friend = scope:daenerys

		# HBO chronology: Jon bends the knee here, not through a betrothal.
		# Only the North follows his oath; the Vale is not transferred.
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = jon_daenerys_fealty_change
			add_claim_on_loss = no
		}
		title:e_the_north.holder ?= {
			change_liege = {
				liege = scope:daenerys
				change = scope:jon_daenerys_fealty_change
			}
		}
		resolve_title_and_vassal_change = scope:jon_daenerys_fealty_change

		remove_variable = agot_pwl_direct
		remove_variable = agot_pwl_top

		# Preserve the established Mega War integration without making Jon
		# independent again after his voluntary oath.
		scope:daenerys = {
			random_owned_story = {
				limit = {
					story_type = story_agot_mw_rebel
					story_owner = scope:daenerys
				}
				save_scope_as = jon_daenerys_mw_rebel_story
			}
			random_character_war = {
				limit = {
					using_cb = atfw_daenerys_conquest_war
					primary_attacker = scope:daenerys
					primary_defender = title:h_the_iron_throne.holder
				}
				save_scope_as = jon_daenerys_conquest_war
			}
		}

		if = {
			limit = {
				exists = scope:jon_daenerys_mw_rebel_story
				exists = scope:jon_daenerys_conquest_war
			}
			scope:jon_daenerys_mw_rebel_story.var:mw_crown_story_var ?= {
				save_scope_as = jon_daenerys_mw_crown_story
			}

			scope:jon_daenerys_mw_crown_story ?= {
				remove_list_variable = { name = mw_neutral_list target = root }
				remove_list_variable = { name = mw_loyalist_list target = root }
			}
			scope:jon_daenerys_mw_rebel_story = {
				add_to_variable_list = { name = mw_rebel_supporter_list target = root }
			}

			every_vassal_or_below = {
				limit = {
					highest_held_title_tier > tier_barony
					agot_is_high_septon = no
					NOT = { primary_title = title:d_kingsguard }
				}
				save_scope_as = northern_vassal
				scope:jon_daenerys_mw_crown_story ?= {
					remove_list_variable = { name = mw_neutral_list target = scope:northern_vassal }
					remove_list_variable = { name = mw_loyalist_list target = scope:northern_vassal }
				}
				scope:jon_daenerys_mw_rebel_story = {
					add_to_variable_list = { name = mw_rebel_supporter_list target = scope:northern_vassal }
				}
			}

			scope:jon_daenerys_conquest_war = { add_attacker = root }
		}

		trigger_event = { id = atfw_jon_daenerys.0016 days = 5 }
	}
}
