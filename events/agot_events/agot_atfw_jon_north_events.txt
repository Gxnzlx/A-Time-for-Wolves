namespace = atfw_jon_north

# Part I of Jon Snow's 8303 storyline.
# This chain is deliberately Jon-only. It establishes the immediate
# post-Battle-of-the-Bastards situation before the existing Dragonstone journey.

# Bran reaches Winterfell before the northern council.
atfw_jon_north.0001 = {
	type = character_event
	title = atfw_jon_north.0001.t
	desc = atfw_jon_north.0001.desc
	theme = family
	override_background = {
		reference = agot_pk_north
	}

	immediate = {
		# Stark_Dummy_10 now represents Bran in the active timeline. Detach the
		# vanished original so only the continuation appears among Ned and
		# Catelyn's children in the family tree.
		character:Stark_10 = {
			set_father = character:Unused_1
			set_mother = character:Unused_2
		}

		character:Stark_Dummy_10 = {
			change_first_name = Brandon
			set_father = character:Stark_3
			set_mother = character:Tully_4
			set_house = character:Stark_3.house
			set_location = root.capital_province
			save_scope_as = bran
		}
	}

	left_portrait = {
		character = root
		animation = shock
	}

	right_portrait = {
		character = scope:bran
		animation = personality_rational
	}

	option = {
		name = atfw_jon_north.0001.a
		trigger_event = { id = atfw_jon_north.0002 days = 10 }
	}
}

# A private gathering reveals Jon's true parentage.
# This adapts CK3AGOT's existing disputed-heritage/royal-bastard secret
# mechanics without publicly exposing the secret or changing Jon's public house.
atfw_jon_north.0005 = {
	type = character_event
	title = atfw_jon_north.0005.t
	desc = atfw_jon_north.0005.desc
	theme = family
	override_background = study

	immediate = {
		character:Stark_Dummy_8 = { save_scope_as = sansa }
		character:Stark_Dummy_10 = { save_scope_as = bran }
		character:Tarly_2 = { save_scope_as = sam }
		character:Reed_1 = { save_scope_as = howland }
		character:Reed_2 = { save_scope_as = meera }

		# Jon's real parents are already Rhaegar Targaryen and Lyanna Stark in
		# CK3AGOT's base history. Reveal the existing disputed-heritage secret
		# to Jon privately, using the same secret type as the base bastard events.
		if = {
			limit = {
				knows_self_royal_bastard_secret = no
			}
			scope:howland = {
				random_known_secret = {
					limit = {
						secret_type = secret_agot_disputed_heritage
						secret_target = root
					}
					reveal_to = root
				}
			}
		}

		# Safety fallback for ATFW timelines where Howland's inherited secret
		# was not instantiated. This remains private to Jon.
		if = {
			limit = {
				knows_self_royal_bastard_secret = no
			}
			add_secret = {
				type = secret_agot_disputed_heritage
				target = root
			}
		}

		add_character_flag = atfw_jon_knows_true_parentage
	}

	left_portrait = {
		character = scope:bran
		animation = personality_rational
	}

	right_portrait = {
		character = scope:sansa
		animation = shock
	}

	lower_left_portrait = {
		character = scope:sam
		animation = personality_honorable
	}

	lower_center_portrait = {
		character = scope:howland
		animation = personality_rational
	}

	lower_right_portrait = {
		character = scope:meera
		animation = personality_bold
	}

	option = {
		name = atfw_jon_north.0005.a

		# Jon learns the truth but refuses to expose it. He remains publicly
		# Jon Snow/Stark as defined by ATFW; no public bastard-reveal story fires.
		add_character_flag = atfw_jon_hides_true_parentage

		trigger_event = { id = atfw_jon_north.0006 days = 12 }
	}
}
