namespace = atfw_marriage_events

atfw_marriage_events.0001 = {
	type = character_event
	title = atfw_marriage_events.0001.t
	desc = atfw_marriage_events.0001.desc
	theme = marriage

	trigger = {
		is_alive = yes
		is_married = no
		is_betrothed = no
	}

	immediate = {
		character:Reed_2 = {
			save_scope_as = Reed_2_scope
		}
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:Reed_2_scope
		animation = personality_bold
	}

	option = { # Accept marriage with Meera Reed
		name = atfw_marriage_events.0001.a

		trigger = {
			scope:Reed_2_scope = {
				is_alive = yes
				is_married = no
				is_betrothed = no
			}
		}

		scope:Reed_2_scope = {
			marry = root
		}

		ai_chance = {
			base = 100
		}
	}

	option = { # Decline marriage
		name = atfw_marriage_events.0001.b

		ai_chance = {
			base = 0
		}
	}
}
