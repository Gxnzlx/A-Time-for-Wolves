atfw_north_journey.0005 = {
 type = character_event
 hidden = yes
 trigger = { this = character:Targaryen_9 location = title:b_winterfell.title_province NOT = { has_character_flag = atfw_north_trip_arrived } }
 immediate = {
  debug_log = "ATFW: shared journey arrived at Winterfell"
  add_character_flag = atfw_north_trip_arrived
  remove_character_flag = atfw_north_trip_active
  remove_character_flag = atfw_north_trip_ready
  # Keep the scripted principals and mounts at the destination after disbanding the entourage.
  set_location = title:b_winterfell.title_province
  character:Stark_7 = {
   set_location = title:b_winterfell.title_province
   trigger_event = { id = atfw_weekly.1045 days = 7 }
  }
  character:dragon_rhaegal ?= { if = { limit = { is_alive = yes } set_location = title:b_winterfell.title_province } }
  character:dragon_drogon ?= { if = { limit = { is_alive = yes } set_location = title:b_winterfell.title_province } }
 }
}
