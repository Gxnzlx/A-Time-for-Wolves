# Serializes the visible Jon/Daenerys storyline globally. Original scopes are carried by trigger_event.

atfw_weekly.1001 = {
 type = character_event
 hidden = yes
 trigger = { is_alive = yes NOT = { has_character_flag = atfw_weekly_shown_atfw_daenerys_pov_0001 } }
 immediate = {
  if = {
   limit = { OR = { has_global_variable = atfw_story_weekly_cooldown atfw_highgarden_story_pending = yes } }
   trigger_event = { id = atfw_weekly.1001 days = 1 }
  }
  else_if = {
   limit = { 
  this = character:Targaryen_9
  is_alive = yes
  NOT = { has_character_flag = atfw_daenerys_pov_0001_seen }
  character:Stark_7 = { is_alive = yes has_character_flag = atfw_daenerys_pov_0001_queued }
  }
   add_character_flag = atfw_weekly_shown_atfw_daenerys_pov_0001
   set_global_variable = { name = atfw_story_weekly_cooldown value = yes days = 7 }
   trigger_event = atfw_daenerys_pov.0001
  }
 }
}
